# Tidy both models
tidy_m1 <- tidy(model_no_season, exponentiate = TRUE, conf.int = TRUE) %>% 
  mutate(model = "no_season")
tidy_m2 <- tidy(model, exponentiate = TRUE, conf.int = TRUE) %>% 
  mutate(model = "season")

# Wide format: one row per term
comparison_wide <- bind_rows(tidy_m1, tidy_m2) %>%
  select(model, term, estimate, conf.low, conf.high, p.value) %>%
  pivot_wider(
    names_from  = model,
    values_from = c(estimate, conf.low, conf.high, p.value)
  ) %>%
  mutate(
    term = recode(term,
      "(Intercept)"  = "Intercept",
      "PC1"          = "PC1",
      "PC2"          = "PC2",
      "PC3"          = "PC3",
      "seasonSpring" = "Spring",
      "seasonSummer" = "Summer",
      "seasonFall"   = "Fall"
    )
  )

# add sig columns to comparison_wide before gt
comparison_wide <- comparison_wide %>%
  mutate(
    sig_no_season = !is.na(p.value_no_season) & p.value_no_season < 0.05,
    sig_season    = !is.na(p.value_season)    & p.value_season    < 0.05
  )

comparison_wide %>%
  gt() %>%
  tab_header(
    title    = "Air Quality, Pollen, and Precipitation Components as Predictors of Daily Asthma Emergency Room Visits: Unadjusted and Season-Adjusted Poisson Regression Models",
    subtitle = "Outcome: Total daily asthma ER visits | Estimates expressed as Incidence Rate Ratios (IRR)"
  ) %>%
  cols_label(
    term                = "Term",
    estimate_no_season  = "IRR",
    conf.low_no_season  = "95% CI (low)",
    conf.high_no_season = "95% CI (high)",
    p.value_no_season   = "p-value",
    estimate_season     = "IRR",
    conf.low_season     = "95% CI (low)",
    conf.high_season    = "95% CI (high)",
    p.value_season      = "p-value"
  ) %>%
  cols_hide(c(sig_no_season, sig_season)) %>%
  tab_spanner(label = "Model 1: No Season",
              columns = c(estimate_no_season, conf.low_no_season, conf.high_no_season, p.value_no_season)) %>%
  tab_spanner(label = "Model 2: Season Adjusted",
              columns = c(estimate_season, conf.low_season, conf.high_season, p.value_season)) %>%
  fmt_number(

                 estimate_season, conf.low_season, conf.high_season),

    columns = c(p.value_no_season, p.value_season),

# tidy and keep only PC rows
tidy_model <- function(mod, label) {
  tidy(mod, exponentiate = TRUE, conf.int = TRUE) %>%
    dplyr::filter(term %in% c("PC1", "PC2", "PC3")) %>%
    dplyr::select(term, estimate, conf.low, conf.high, p.value) %>%
    mutate(model = label)
}

sensitivity_long <- bind_rows(
  tidy_model(model,       "Season"),
  tidy_model(model_temp,  "Temperature"),
  tidy_model(model_dow,   "Day of Week"),
  tidy_model(model_month, "Month")
)

# helper to format p-values
fmt_p <- function(x) {
  case_when(
    is.na(x)  ~ "—",
    x < 0.001 ~ "<0.001",
    TRUE       ~ formatC(x, digits = 3, format = "f")
  )
}
